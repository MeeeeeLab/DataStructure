```java
import java.util.NoSuchElementException;

public class StackChapterCheck {
    private static class Node {
        char value;
        Node next;
        Node(char value, Node next) { this.value = value; this.next = next; }
    }

    private Node top;
    private int size;

    void push(char value) { top = new Node(value, top); size++; }

    char pop() {
        if (top == null) throw new NoSuchElementException("stack is empty");
        char value = top.value;
        top = top.next;
        size--;
        return value;
    }

    char peek() {
        if (top == null) throw new NoSuchElementException("stack is empty");
        return top.value;
    }

    boolean isEmpty() { return top == null; }
    int size() { return size; }

    static boolean isBalanced(String text) {
        StackChapterCheck stack = new StackChapterCheck();
        for (char ch : text.toCharArray()) {
            if (ch == '(' || ch == '[' || ch == '{') stack.push(ch);
            else if (ch == ')' || ch == ']' || ch == '}') {
                if (stack.isEmpty()) return false;
                char left = stack.pop();
                if (!(left == '(' && ch == ')' || left == '[' && ch == ']'
                        || left == '{' && ch == '}')) return false;
            }
        }
        return stack.isEmpty();
    }

    static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    public static void main(String[] args) {
        StackChapterCheck stack = new StackChapterCheck();
        stack.push('A'); stack.push('B');
        check(stack.peek() == 'B' && stack.size() == 2, "peek/size");
        check(stack.pop() == 'B' && stack.pop() == 'A' && stack.isEmpty(), "LIFO");
        try {
            stack.pop();
            throw new AssertionError("underflow accepted");
        } catch (NoSuchElementException expected) { }
        check(isBalanced("{[()]}"), "balanced");
        check(!isBalanced("{[(])}"), "mismatch");
        check(isBalanced(""), "empty input");
        System.out.println("stack-chapter checks: OK");
    }
}
```
